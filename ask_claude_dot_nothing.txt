```python
import os
import re

def search_word():
    # Ask the user to input a word
    word = input("Enter a word or phrase to search for: ")

    # Get the full file paths
    files = []

    # Loop through all the directories and files, appending names of files
    # that contain the given filename (case-insensitive)
    for root, dirs, files in os.walk('path_to_your_directory/'):
        for file in files:
            if re.search(word, file):
                files.append(os.path.join(root, file))

    # If there are any matches, print them out
    if len(files) > 0:
        print("\nResults:")
        for file in files:
            with open(file, 'r') as f:
                content = f.read()
                for pattern in re.findall(word, content):
                    print(f"Found match: {pattern} in {file}")
        print("\n")

    else:
        print("No matches found.")

search_word()
```
